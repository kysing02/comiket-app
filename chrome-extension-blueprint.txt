// content.js
function scrapeTweets() {
  const tweets = document.querySelectorAll('article[data-testid="tweet"]');
  const results = [];
  
  tweets.forEach(tweet => {
    const text = tweet.querySelector('div[data-testid="tweetText"]')?.innerText;
    const images = Array.from(tweet.querySelectorAll('img[src*="pbs.twimg.com/media"]')).map(img => img.src);
    const url = tweet.querySelector('a[href*="/status/"]')?.href;
    const postId = url?.split('/').pop();
    
    if (text && url) {
      results.push({ post_id: postId, url, content: text, image_urls: images });
    }
  });
  
  return results;
}
